# Darwin SDK _time.h: CLOCK_MONOTONIC=6. Use the SAME clock as the native
# collector/hook, rather than directly comparing Python mach absolute time.
class Timespec(ctypes.Structure):
 _fields_=[('sec',ctypes.c_long),('nsec',ctypes.c_long)]
libsystem=ctypes.CDLL('/usr/lib/libSystem.B.dylib',use_errno=True)
libsystem.clock_gettime.argtypes=[ctypes.c_int,ctypes.POINTER(Timespec)]
libsystem.clock_gettime.restype=ctypes.c_int
def native_ns():
 stamp=Timespec()
 if libsystem.clock_gettime(6,ctypes.byref(stamp))!=0:raise OSError(ctypes.get_errno(),'clock_gettime CLOCK_MONOTONIC')
 return stamp.sec*1000000000+stamp.nsec
