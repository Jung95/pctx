use std::{fs,io::Read,process::{Command,Stdio},os::unix::{fs::PermissionsExt,process::CommandExt},sync::{Arc,Barrier},time::{Duration,Instant}};
fn main(){let base=std::env::temp_dir().join(format!("pctx-startup-isolation-{}",std::process::id()));fs::create_dir(&base).unwrap();
for mode in ["interpreter-group","script-group","script-no-group","interpreter-no-group"] {let barrier=Arc::new(Barrier::new(8));let started=Instant::now();let mut rows=vec![];std::thread::scope(|scope|{let mut handles=vec![];for worker in 0..8 {let b=barrier.clone();let dir=base.join(format!("{mode}-{worker}"));fs::create_dir(&dir).unwrap();let executable=dir.join("git");fs::write(&executable,"#!/bin/sh\nprintf stdout\nprintf stderr >&2\n").unwrap();fs::set_permissions(&executable,fs::Permissions::from_mode(0o700)).unwrap();handles.push(scope.spawn(move||{let mut outcomes=vec![];b.wait();for iteration in 0..16 {let mut c=if mode.starts_with("interpreter-") {let mut c=Command::new("/bin/sh");c.arg(&executable);c}else{Command::new(&executable)};c.args(["rev-parse","--is-inside-work-tree"]).current_dir(&dir).env_clear().stdin(Stdio::null()).stdout(Stdio::piped()).stderr(Stdio::piped());if !mode.ends_with("no-group"){c.process_group(0);}let began=Instant::now();let mut child=c.spawn().unwrap();let spawn_ms=began.elapsed().as_micros();let pid=child.id();let mut expired=false;let status=loop{if began.elapsed()>=Duration::from_millis(1000){expired=true;let _=child.kill();break child.wait().unwrap();}if let Some(status)=child.try_wait().unwrap(){break status;}std::thread::sleep(Duration::from_millis(2));};let mut out=vec![];let mut err=vec![];child.stdout.take().unwrap().read_to_end(&mut out).unwrap();child.stderr.take().unwrap().read_to_end(&mut err).unwrap();let ok=!expired&&status.success()&&out==b"stdout"&&err==b"stderr";outcomes.push(format!("{{\"worker\":{worker},\"iteration\":{iteration},\"pid\":{pid},\"spawn_us\":{spawn_ms},\"expired\":{expired},\"ok\":{ok},\"stdout_bytes\":{},\"stderr_bytes\":{},\"elapsed_ms\":{}}}",out.len(),err.len(),began.elapsed().as_millis()));if !ok{break;}}outcomes}));}for h in handles{rows.extend(h.join().unwrap());}});println!("{{\"mode\":\"{mode}\",\"elapsed_ms\":{},\"attempts\":[{}]}}",started.elapsed().as_millis(),rows.join(","));}
fs::remove_dir_all(base).unwrap();}
