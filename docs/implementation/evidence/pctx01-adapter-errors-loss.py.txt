import hashlib,json,os,pathlib,subprocess
root=pathlib.Path("/Users/dev/PCTX");path=root/"src/adapter.rs";original=path.read_bytes();source=original.decode()
env=dict(os.environ,RUSTUP_HOME=str(root/".toolchain/rustup"),CARGO_HOME=str(root/".toolchain/cargo"),SDKROOT="/Library/Developer/CommandLineTools/SDKs/MacOSX.sdk",PATH=str(root/".toolchain/cargo/bin")+":/Library/Developer/CommandLineTools/usr/bin:/opt/homebrew/bin:/usr/bin:/bin:/usr/sbin:/sbin")
prefix=root/"docs/implementation/evidence/pctx01-adapter-errors"
helper=source[source.index("fn err("):source.index("fn label(")]
cases=[
("blanket",helper,'fn err(code: &str, s: &str) -> Error { Error::new(code, s, if code == "INVALID_ARGUMENT" { 2 } else { 9 }) }\n',"native_adapter_owner_refusal_is_policy_five_in_both_formats"),
("config",'Error::new("CONFIG_CONFLICT", "Existing hooks must be an object", 2)','err("CONFIG_CONFLICT", "Existing hooks must be an object")',"native_adapter_input_cap_and_configuration_shape_are_input_two"),
("plan-source",'Error::new("PLAN_MISMATCH", "Plan bytes changed", 4)','err("PLAN_MISMATCH", "Plan bytes changed")',"direct_adapter_plan_input_source_and_binding_are_distinct"),
("state",'        _ => 9,','        "IDEMPOTENCY_CONFLICT" => 4,\n        _ => 9,',"direct_adapter_receipt_conflicts_and_reconciliation_retain_nine"),
("schema",'"DB_SCHEMA_TOO_NEW" => 7,','"DB_SCHEMA_TOO_NEW" => 9,',"direct_adapter_schema_and_ownership_integrity_are_storage_seven")]
results=[]
try:
 for name,old,new,test in cases:
  assert source.count(old)==1,(name,source.count(old))
  path.write_text(source.replace(old,new,1))
  log=pathlib.Path(str(prefix)+"-loss-"+name+".log")
  with log.open("wb") as output:
   run=subprocess.run(["cargo","test","--locked","--test","adapter_errors",test,"--","--exact","--nocapture"],cwd=root,env=env,stdout=output,stderr=subprocess.STDOUT)
  data=log.read_text();results.append(dict(case=name,test=test,exit=run.returncode,actual_test_failed="test result: FAILED. 0 passed; 1 failed;" in data))
  assert run.returncode==101 and results[-1]["actual_test_failed"],results[-1]
finally:
 path.write_bytes(original)
 restored=path.read_bytes()==original
 pathlib.Path(str(prefix)+"-loss-results.json").write_text(json.dumps(dict(results=results,exact_source_restored=restored,source_sha256=hashlib.sha256(original).hexdigest()),indent=2)+"\n")
 assert restored
print(json.dumps(results))
