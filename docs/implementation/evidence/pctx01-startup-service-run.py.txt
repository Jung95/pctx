def digest(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def wait_marker(root,name,auth,seconds):
    end=time.monotonic()+seconds if seconds is not None else None
    path=root/name
    while not path.exists():
        if auth.poll() is not None:raise RuntimeError('collector exited before marker')
        if end is not None and time.monotonic()>=end:raise RuntimeError('marker wait expired')
        time.sleep(.02)
    handle=os.open(str(path),os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)

        info=os.fstat(stream.fileno())
        if not stat.S_ISREG(info.st_mode) or info.st_uid!=0 or info.st_size>8192:
            raise RuntimeError('root marker refused')
        return json.loads(stream.read(8193))
