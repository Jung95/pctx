import hashlib,json,pathlib,shlex,subprocess
p=pathlib.Path("src/render.rs")
original=p.read_bytes()
e=pathlib.Path("docs/implementation/evidence")
cases=[("child", b'"child_exit_code",', b'"omitted_child_exit_code",', "actual_child_codes_and_signal_are_distinct_from_wrapper_status"),("refusal", b"if preserve_refusal {", b"if false {", "actual_prelaunch_refusal_has_no_child_outcome_or_invocation")]
results=[]
try:
 for name,old,new,test in cases:
  assert original.count(old)==1
  p.write_bytes(original.replace(old,new))
  with (e/("pctx01-budget-fallback-loss-"+name+".log")).open("w") as f:
   r=subprocess.run(shlex.split("env RUSTUP_HOME=/Users/dev/PCTX/.toolchain/rustup CARGO_HOME=/Users/dev/PCTX/.toolchain/cargo SDKROOT=/Library/Developer/CommandLineTools/SDKs/MacOSX.sdk PATH=/Users/dev/PCTX/.toolchain/cargo/bin:/Library/Developer/CommandLineTools/usr/bin:/opt/homebrew/bin:/usr/bin:/bin:/usr/sbin:/sbin ")+["cargo","test","--locked","--test","cli_run_exit",test,"--","--exact"],stdout=f,stderr=subprocess.STDOUT,timeout=40)
  results.append({"mutation":name,"test_exit":r.returncode})
  assert r.returncode==101
finally:
 p.write_bytes(original)
restored=hashlib.sha256(p.read_bytes()).hexdigest()
(e/"pctx01-budget-fallback-loss-results.json").write_text(json.dumps({"results":results,"original_sha256":hashlib.sha256(original).hexdigest(),"restored_sha256":restored},indent=2)+"\n")
print(results,"exact restore",original==p.read_bytes())
