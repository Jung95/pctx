def admit(row):
    if not isinstance(row,dict) or row.get('processID')!=SERVICE_PID or row.get('processImagePath')!=SOURCE or row.get('eventType')!='signpostEvent' or row.get('signpostScope')!='process':return None
    try:

        if stamp.tzinfo is None or not START<=stamp.astimezone(datetime.timezone.utc)<=END:return None
    except (ValueError,TypeError):return None
    ident=row.get('signpostID')
    if type(ident)!=int or not 0<ident<2**64-2:return None
    return stamp.astimezone(datetime.timezone.utc),ident

def path_ids(rows,expected):
    if not isinstance(rows,list) or len(rows)>16:raise ValueError('path event cap')
    matched=[]
    for row in rows:

        if admitted is None:continue
        stamp,ident=admitted

    return matched
